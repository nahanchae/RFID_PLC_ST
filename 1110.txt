(*
RFID 수집 프로그램 인수인계 설명

[처리 순서]
0 시작 대기 -> 10 Inventory 송신 완료 대기 -> 11 재송신 간격 대기
-> 10 반복 -> 20 수신 누적 -> 30 Stop 요청 -> 40 Stop 송신 완료 대기
-> 50 메시지 분석/집계 -> 60 TAG 변환/선정 TAG 선정 -> 90 완료 -> 0 대기.
CASE는 한 스캔에 진입 시점 State의 한 분기만 실행한다.
분기 안에서 State를 바꾸면 새 상태의 분기는 다음 스캔부터 실행한다.

[첨부 코드의 현재 설정]
Inventory 총 3회, 송신 완료 후 재송신까지 100ms, 수집 3초, 송신 대기 제한 6초.
수집 타이머는 State 20에서만 동작한다. Inventory 재송신 대기 시간은 수집 3초에 포함되지 않는다.
현재 수신 데이터 누적도 State 20에서만 수행한다.
Inventory 송신 완료는 PLC 소켓 명령의 정상 완료이며 리더의 명령 수락 ACK 확인은 아니다.
실제 소켓 접속, 송신 버퍼 준비, GP.SOCSND/GP.SOCRCV 실행은 외부 래더에서 처리한다.

[사용 메모리]
R11790.0: Start 입력 / R11790.2: Stop 입력.
D6000: 소켓 수신 바이트 수 / D6001 이후: 한 WORD에 2바이트인 수신 본문.
D4000~D5999: 수신 본문을 한 WORD에 1바이트씩 누적, 총 2000바이트 용량.
R12100~R12103: TAG1 8문자 / R12104~R12105: TAG1 DINT 반복횟수.
R12106~R12109: TAG2 8문자 / R12110~R12111: TAG2 DINT 반복횟수.
R12112~R12115: TAG3 8문자 / R12116~R12117: TAG3 DINT 반복횟수.
R12118: 서로 다른 등록 TAG 개수 / R12119: 모드 입력(0 COUNT, 1 최고 RSSI, 2 COUNT+RSSI 혼합). Start에서 저장하며 초기화하지 않음.
R12120~R12123: 선정 TAG 8문자 / R12124~R12125: 그 TAG의 DINT 반복횟수.
R12126~R12128: TAG1~3 최고 RSSI, signed INT, 0.1 dBm 단위.
R12129: 선정 TAG 최고 RSSI, signed INT, 0.1 dBm 단위. R12133=1일 때 유효.
R12130~R12132: TAG1~3 RSSI 유효 여부(0/1).
R12133: 선정 TAG RSSI 유효(0/1). R12134: RSSI 누락/형식 경고(0/1).
R12135: RSSI 선정 불가 오류(0/1). R12136: 모드 설정 오류(0/1).
R12137: START에서 저장한 실제 모드.
R12138: 허용 COUNT 차이(WORD 0~65535), 예: 4. START 전 외부에서 설정하며 ST에서 지우지 않음.
R12139: 혼합 모드에서 RSSI 부족으로 COUNT 결과 사용(0/1).
혼합 후보: 최다 COUNT - 해당 TAG COUNT <= CountGapLimit.
혼합 후보 중 하나라도 유효 RSSI가 없으면 최다 COUNT 결과를 사용한다.
R12140~R12143: HEXA용 RSSI ASCII 임시 버퍼, 한 WORD에 문자 하나.
R12144~R12147: HEXA 출력 예약 영역. 4문자 결과는 R12144의 한 WORD.
SM8161: HEXA 실행 직전에 TRUE로 설정하고 직후 기존 상태로 복원.
인터럽트 프로그램에서 HEXA/ASCI/CRC/CCD를 사용한다면 그쪽도 실행 전 모드 설정 필요.
HEXA는 ASCII 0~9/A~F만 허용. 소문자는 대문자로 바꾸고 형식 검사 후 실행한다.
유효 RSSI를 읽을 때 TAG별 최고값을 저장한다. 예: -486 -> -48.6 dBm.
COUNT 모드에서도 선정 TAG의 최고 RSSI를 표시한다. RSSI가 없으면 ResultRssiValid=FALSE.
RSSI 모드는 유효 RSSI 1개 이상인 TAG만 후보. 자동 COUNT 복귀 없음.
최고 RSSI 동률이면 먼저 등록된 TAG를 유지한다. COUNT 동률도 먼저 등록된 TAG를 유지한다.
RSSI 필드: 23~24번째 ;R, 25~28번째 signed 16bit HEX. CR/LF로 종료.
COUNT 모드는 RSSI 없는 기존 메시지도 집계한다.
RSSI 누락/형식 오류 응답은 COUNT만 집계. 다른 유효 응답은 최고 RSSI 비교에 사용.
RSSI 모드에서 TAG는 있으나 유효 RSSI가 전혀 없으면 ErrRssiMissing으로 결과 무효.
TAG1~3은 처음 등록된 순서이며 COUNT 내림차순 정렬 결과가 아니다.
문자열 출력은 NUL 종료값 없이 고정 길이 8문자로 읽는다.
D3000: 송신 종류(1 Inventory, 2 Stop).
D3001.0: 송신 요청 / D3001.1: 완료 및 유효 결과 / D3001.2: 처리 오류.
Z15: 수신 원본 WORD 주소 또는 선정 TAG 복사 원본 주소의 인덱스.
Z16: 누적 저장/분석 주소의 인덱스 / Z17: TAG 및 반복횟수 출력 주소의 인덱스.

[결과 신호 해석]
Done: 이번 처리가 끝났음. TAG 미인식이나 오류 종료에도 TRUE가 될 수 있다.
ResultValid: 선정 TAG가 선정됐고 Error가 없음. Overflow는 이 판정에서 제외한다.
TagReadFail: 완료 시 DataTotal=0이고 Error가 없음. 4FC0 필터를 통과한 TAG가 없다는 뜻.
Error: 통신, 송수신 대기 제한, 길이/인덱스 이상 등의 처리 오류.
Overflow: 버퍼 용량 또는 TAG 등록 개수 초과 경고. 저장된 범위의 결과는 사용한다.
버퍼 초과 시 이번 수신분 전체를 버리고 Stop을 요청한다. 첫 수신분부터 초과하면 저장 TAG가 없을 수 있다.
4번째 신규 TAG는 등록하지 않지만 기존 3개 TAG의 반복횟수는 계속 집계한다.
최다 결과는 저장/등록된 데이터 범위에서의 최다이며 전체 응답의 최다를 보장하지 않는다.
TagTextError는 변환/표시 진단이며 현재 Error 또는 ResultValid 판정에 포함되지 않는다.
Done, ResultValid, 오류/경고 이력 및 결과 영역은 다음 정상 Start 접수 시 초기화한다.
R12119 설정은 초기화하지 않는다. 외부 래더/HMI에서 0 또는 1로 설정한다.
*)

(* 추가 라벨: TagReadFail / Bit (BOOL) / VAR
의미: 오류 없이 완료됐으나 4FC0 필터를 통과한 TAG 없음.
Overflow는 경고: 저장분 기준 결과 사용 가능. 전체 응답의 선정 TAG는 보장하지 않음.
D3001.0: 송신 요청 / D3001.1: Done AND ResultValid
*)


//Start1 상승 에지

// StartRise(Start);
// StartRise := StartEdge.Q;  이런식으로 함수호출해서 사용해도 됨
// Start가 참이고 start prev가 거짓일때만 동작함, start가 참으로 유지되더라도 prev로 인해 동작하지 않는다
Start1 := R11790.0;
(* 이전 스캔은 OFF, 현재 스캔은 ON인 순간만 TRUE. State 0에서만 새 작업을 접수한다. *)
StartRise := Start1 AND NOT StartPrev;
(* 다음 스캔 비교를 위해 현재 Start 상태를 기억한다. *)
StartPrev := Start1;

//송신 요청은 한 스캔 펄스
(* 스캔 시작 시 요청 해제. 아래 CASE에서 송신이 필요할 때만 TRUE로 하여 한 스캔 요청을 만든다. *)
TxRequest := FALSE;

//정상 완료 / 이상 완료
(* 완료 비트 ON + 이상 비트 OFF를 정상 완료로 판단. 변수명 Pulse 자체가 에지 검출을 하지는 않는다. *)
TxDonePulse  := R10200.2 AND NOT R10200.3;
TxErrorPulse := R10200.2 AND R10200.3;

(* 외부 수신 명령의 완료/이상 비트를 조합한다. 완료 신호가 연속 유지되면 중복 처리할 수 있으므로 외부 래더의 완료 신호 처리가 필요하다. *)
RxDonePulse  := R10200.4 AND NOT R10200.5; // 응답 확인 후 D4000번지로 MOV한다
RxErrorPulse := R10200.4 AND R10200.5;

//수집 시간
(* TON은 매 스캔 호출한다. IN이 FALSE면 비유지형 타이머가 리셋되고, 설정시간 경과 시 Q가 TRUE가 된다. *)
ReadTimer(
IN := (State = 20),
PT := T#3s
);

//송신 완료 대기 제한
SendTimer(
IN := ((State = 10) OR (State = 40)),
PT := T#6s
);

//Inventory 재송신 간격
InventoryGapTimer(
IN := (State = 11),
PT := T#100ms
);

//송신 대기 및 수집 중 Stop 입력 기억
Stop1 := R11790.2;
IF ((State = 10)
    OR (State = 11)
    OR (State = 20)) AND Stop1 THEN

    (* Stop 입력이 짧게 들어와도 다음 상태에서 정지 요청을 잊지 않도록 기억한다. *)
    StopLatched := TRUE;
    StoppedByUser := TRUE;

END_IF;

//통신 진행 중 발생한 수신 오류 기억
IF ((State = 10)
    OR (State = 11)
    OR (State = 20)
    OR (State = 30)
    OR (State = 40)) AND RxErrorPulse THEN

    (* CASE보다 먼저 오류를 기록한다. 바로 뒤 Error 대입은 같은 스캔의 CASE에서도 오류를 검사하기 위한 것이다. *)
    ErrReceive := TRUE;
    Error := TRUE;

END_IF;


//=========================================================
//상태 제어
//=========================================================

CASE State OF

    //-----------------------------------------------------
    //0: 시작 대기
    //-----------------------------------------------------
    0:
    IF StartRise AND NOT Stop1 THEN

        Done := FALSE;
        TagReadFail := FALSE; (* 새 Start에서 미인식 상태 초기화 *)
        // 새 측정 시작 시에만 원인별 오류 이력 초기화
        ErrComm := FALSE;
        ErrInventoryTimeout := FALSE;
        ErrInventorySend := FALSE;
        ErrReceive := FALSE;
        ErrStopTimeout := FALSE;
        ErrStopSend := FALSE;
        ErrRawCount := FALSE;
        ErrRxLength := FALSE;
        ErrBufferFull := FALSE;
        ErrAnalyzeIndex := FALSE;
        ErrDataFull := FALSE;
        ErrState := FALSE;
        ErrSelectMode := FALSE;
        ErrRssiMissing := FALSE;
        WarnRssiData := FALSE;
        ResultRssi := 0;
        ResultRssiValid := FALSE;
        MaxCount := 0;
        CountBestIndex := -1;
        RssiBestIndex := -1;
        CandidatesValid := TRUE;
        HybridFallback := FALSE;
        CountGapLimit := WORD_TO_DINT(R12138); (* 0~65535、START 시 저장 *)
        (* R12119는 외부에서 설정. 이번 수집 중 변경되어도 적용 모드는 유지한다. *)
        ActiveSelectMode := WORD_TO_INT(R12119);
        Error := FALSE;
        Overflow := FALSE;
        StoppedByUser := FALSE;
        ResultValid := FALSE;

        StopLatched := FALSE;
        (* 현재 문자가 메시지 끝을 완성한 경우에만 아래 ParseState 3에서 TRUE가 된다. *)
        KeyReady := FALSE;
        Match4FC0 := FALSE;

        InventorySendCount := 0;

        (* 누적한 바이트 개수이자 다음에 쓸 D4000 기준 오프셋. 새 작업에서 0부터 시작한다. *)
        RawCount := 0;
        (* 누적 데이터에서 다음에 분석할 바이트 위치. 수신 중 ByteIndex와 별도로 관리한다. *)
        AnalyzeIndex := 0;

        ParseState := 0;
        BytePos := 0;
        KeyLength := 0;

        DataTotal := 0;
        (* -1은 일치하는 기존 TAG가 아직 없다는 표시. TAG 배열의 유효 인덱스는 0부터 시작한다. *)
        FoundIndex := -1;
        BestIndex := -1;
        ResultCount := 0;

        (* 새 수집 시작 시 이전 표시 결과 초기화 *)

        (* TAG 공개 영역 + 선정 TAG 출력 영역 *)
        (* 새 Start에서 결과 영역만 초기화. 모드 입력 R12119는 보존 *)
        (* FMOV 인수: 실행조건, 채울 값, WORD 개수, 시작주소 *)
        FMOV(TRUE, 0, 19, R12100); (* R12100~12118 *)
        FMOV(TRUE, 0, 18, R12120); (* R12120~12137, 모드 입력 R12119 제외 *)

        ResultText := '';
        FOR DecodeTag := 0 TO 2 DO
            TagText[DecodeTag] := '';
            CountList[DecodeTag] := 0;
            RssiMax[DecodeTag] := 0;
            RssiSeen[DecodeTag] := FALSE;
            TagTextError[DecodeTag] := FALSE;
        END_FOR;


        FOR kk := 0 TO 15 DO
            KeyCode[kk] := WORD#0;
            ResultCode[kk] := WORD#0;
        END_FOR;

        Z15 := 0;
        Z16 := 0;

        IF (ActiveSelectMode < 0) OR (ActiveSelectMode > 2) THEN
            ErrSelectMode := TRUE;
            Error := TRUE;
            State := 90;
        ELSIF CommError THEN

            ErrComm := TRUE;
            Error := TRUE;
            State := 90;

        ELSE

            (* Inventory 1회차 송신 요청 *)
            (* 최초 송신 요청을 1회차로 센다. 이후 State 11에서 증가시키므로 <3 조건이면 총 3회 요청한다. *)
            InventorySendCount := 1;

            TxCommand := 1;
            TxRequest := TRUE;

            State := 10;

        END_IF;

    END_IF;


    //-----------------------------------------------------
    //10: Inventory 송신 완료 대기
    //-----------------------------------------------------
    10:
    IF CommError OR SendTimer.Q THEN

        (* 진행 중 송신 상태가 불명확하므로 추가 요청 금지 *)
        IF CommError THEN ErrComm := TRUE; END_IF;
        IF SendTimer.Q THEN ErrInventoryTimeout := TRUE; END_IF;
        IF TxErrorPulse THEN ErrInventorySend := TRUE; END_IF;
        Error := TRUE;
        ResultValid := FALSE;
        State := 90;

    ELSIF TxErrorPulse THEN

        (* 이상 완료 후 Stop 송신 시도 *)
        ErrInventorySend := TRUE;
        Error := TRUE;
        State := 30;

    ELSIF TxDonePulse THEN

        IF StopLatched OR Error THEN

            State := 30;

            (* 총 송신 요청 횟수 설정 위치. 4회 전송으로 변경하려면 이 비교값을 4로 수정한다. *)
        ELSIF InventorySendCount < 3 THEN   // INVENTORY 메시지를 3번 전송

            State := 11;

        ELSE

            (* 세 번째 송신 완료 → 3초 수집 시작 *)
            State := 20;

        END_IF;

    END_IF;


    //-----------------------------------------------------
    //11: Inventory 재송신 전 0.1초 대기
    //-----------------------------------------------------
    11:
    IF CommError THEN

        ErrComm := TRUE;
        Error := TRUE;
        State := 90;

    ELSIF StopLatched OR Error THEN

        State := 30;

    ELSIF InventoryGapTimer.Q THEN

        (* 현재 송신 완료 -> State 11의 100ms 경과 -> 다음 회차 송신 요청 순서다. *)
        InventorySendCount := InventorySendCount + 1;

        TxCommand := 1;
        TxRequest := TRUE;

        State := 10;

    END_IF;


    //-----------------------------------------------------
    //20: D6000 수신 데이터를 D4000부터 누적 저장
    //-----------------------------------------------------
    20:
    IF CommError THEN

        ErrComm := TRUE;
        Error := TRUE;
        State := 90;

    ELSIF Error THEN

        State := 30;

    ELSIF StopLatched OR ReadTimer.Q THEN

        (* 정지 조건 우선: 이 스캔부터 추가 저장하지 않음 *)
        State := 30;

    ELSIF RxDonePulse THEN

        (* WORD 수신 길이를 DINT로 변환한다. 이 변환 결과는 0~65535이므로 현재 변환만으로 음수가 되지는 않는다. *)
        RxLength := WORD_TO_DINT(D6000);

        IF (RawCount < 0) OR (RawCount > 2000) THEN

            ErrRawCount := TRUE;
            Error := TRUE;
            State := 30;

        ELSIF (RxLength < 0) OR (RxLength > 2046) THEN

            ErrRxLength := TRUE;
            Error := TRUE;
            State := 30;

            (* 수신분 전체가 남은 공간에 들어가는지 먼저 검사한다. 일부만 복사하지 않고 이번 수신분을 통째로 제외한다. *)
        ELSIF RxLength > (2000 - RawCount) THEN

            //이번 수신분 전체를 담을 공간이 없으면 종료
            Overflow := TRUE;
            ErrBufferFull := TRUE;

            State := 30;

        ELSIF RxLength > 0 THEN

            FOR ByteIndex := 0 TO RxLength - 1 DO

                //D6001부터 수신 WORD 읽기
                (* ByteIndex 0,1은 원본 WORD 0 / 2,3은 원본 WORD 1. 정수 나눗셈으로 원본 WORD 위치를 구한다. *)
                WordIndex := ByteIndex / 2;
                (* 주소 인덱스용 Z는 16비트이므로 INT로 변환. D6001Z15는 D(6001+Z15)를 의미한다. *)
                Z15 := DINT_TO_INT(WordIndex);
                WorkWord := D6001Z15;

                //한 WORD에서 바이트 하나 추출
                (* 0 MOD 2=0이므로 첫 바이트는 짝수 분기. 짝수 위치는 하위 8비트, 홀수 위치는 상위 8비트다. *)
                IF (ByteIndex MOD 2) = 0 THEN

                    (* 상위 8비트를 0으로 만들고 하위 8비트만 남긴다. *)
                    WorkWord := WorkWord AND WORD#16#00FF;

                ELSE
                    // WorkWord: 우측으로 8비트씩 이동시키며 D4000번지에 누적을 위해 사용
                    (* 상위 바이트를 오른쪽 8비트 이동해 하위 바이트 위치로 가져온다. *)
                    WorkWord := SHR(WorkWord, 8);

                END_IF;

                //D4000부터 순서대로 누적
                (* D4000Z16의 실제 저장주소는 D4000+현재 누적 바이트 수. 원본과 달리 한 WORD에 한 바이트씩 저장한다. *)
                Z16 := DINT_TO_INT(RawCount);
                D4000Z16 := WorkWord;
                //RawCount전체 누적 데이터의 다음 저장위치
                RawCount := RawCount + 1;

            END_FOR;

            Z15 := 0;
            Z16 := 0;

        END_IF;

    END_IF;


    //-----------------------------------------------------
    //30: Stop 송신 요청
    //-----------------------------------------------------
    30:
    IF CommError THEN

        ErrComm := TRUE;
        Error := TRUE;
        State := 90;

    ELSE

        TxCommand := 2;
        TxRequest := TRUE;

        State := 40;

    END_IF;


    //-----------------------------------------------------
    //40: Stop 송신 완료 대기
    //-----------------------------------------------------
    40:
    IF CommError OR SendTimer.Q OR TxErrorPulse THEN

        (* 실패해도 저장분은 분석하되 결과는 무효 *)
        IF CommError THEN ErrComm := TRUE; END_IF;
        IF SendTimer.Q THEN ErrStopTimeout := TRUE; END_IF;
        IF TxErrorPulse THEN ErrStopSend := TRUE; END_IF;
        Error := TRUE;

        (* 누적 데이터에서 다음에 분석할 바이트 위치. 수신 중 ByteIndex와 별도로 관리한다. *)
        AnalyzeIndex := 0;
        ParseState := 0;
        BytePos := 0;
        KeyLength := 0;

        State := 50;

    ELSIF TxDonePulse THEN

        (* 누적 데이터에서 다음에 분석할 바이트 위치. 수신 중 ByteIndex와 별도로 관리한다. *)
        AnalyzeIndex := 0;
        ParseState := 0;
        BytePos := 0;
        KeyLength := 0;

        State := 50;

    END_IF;


    //-----------------------------------------------------
    //50: 메시지 분석 → 필터 → 집계
    //-----------------------------------------------------
    50:
    IF (RawCount < 0)
        OR (RawCount > 2000)
        OR (AnalyzeIndex < 0)
        OR (AnalyzeIndex > RawCount) THEN

        IF (RawCount < 0) OR (RawCount > 2000) THEN ErrRawCount := TRUE; END_IF;
        IF (AnalyzeIndex < 0) OR (AnalyzeIndex > RawCount) THEN ErrAnalyzeIndex := TRUE; END_IF;
        Error := TRUE;
        ResultValid := FALSE;
        State := 90;

    ELSE

        (* 스캔당 최대 128바이트 분석 *)
        (* 분석 작업을 여러 스캔에 나눠 수행한다. RawCount 최대 2000이면 바이트 읽기에는 최대 16회의 분석 스캔이 필요하다. *)
        FOR ii := 1 TO 128 DO

            IF AnalyzeIndex >= RawCount THEN
                EXIT;
            END_IF;

            Z16 := DINT_TO_INT(AnalyzeIndex);
            ChCode := D4000Z16;
            //ChCode 문자 하나를 담아서 >T 문자열에서의 TAG정보를 키값에 넣는다

            AnalyzeIndex := AnalyzeIndex + 1;
            (* 현재 문자가 메시지 끝을 완성한 경우에만 아래 ParseState 3에서 TRUE가 된다. *)
            KeyReady := FALSE;


            //=========================================================
            //1. 메시지에서 7~22번째 바이트 추출, >T????7890123456789012(h0D)(h0A)
            //=========================================================

            (* ASCII 003E는 '>'. 기존 분석 상태와 관계없이 새 메시지의 시작으로 동기화한다. *)
            IF ChCode = WORD#16#003E THEN

                (* '>' : 새 메시지 시작 *)
                ParseState := 1;
                BytePos := 1;
                RssiFieldOK := FALSE;
                RssiDigits := 0;
                RssiCharValid := FALSE;
                RssiHexOk := FALSE;
                FMOV(TRUE, 0, 8, R12140); (* HEXA 입력 4워드 + 출력 예약 4워드 초기화 *)
                RssiValue := 0;
                RssiMessageValid := FALSE;
                KeyLength := 0;

            ELSE

                (* ParseState 0: '>' 대기 / 1: 'T' 확인 / 2: 본문 추출 / 3: CR 다음 LF 확인. 0은 아무 동작 없이 다음 문자를 기다린다. *)
                CASE ParseState OF

                    //0: 주석만 존재할 경우 에러 발생함
                    //	'>'

                    1:
                    // 'T'
                    IF ChCode = WORD#16#0054 THEN

                        BytePos := 2;
                        ParseState := 2;

                    ELSE

                        ParseState := 0;

                    END_IF;

                    2:
                    IF ChCode = WORD#16#000D THEN

                        (* CR *)
                        ParseState := 3;

                    ELSIF ChCode = WORD#16#000A THEN

                        (* CR 없는 LF는 폐기 *)
                        ParseState := 0;

                    ELSE

                        (* BytePos는 응답 내 1부터 시작하는 문자 위치. 29 이후는 증가를 멈춰 뒤쪽 문자가 TAG 구간에 다시 들어오지 않게 한다. *)
                        IF BytePos < 29 THEN
                            BytePos := BytePos + 1;
                        END_IF;

                        (*BytePos현재 검색중인 문자열에서의 위치*)
                        IF (BytePos >= 7)
                            AND (BytePos <= 22)
                            AND (KeyLength < 16) THEN

                            (* 본문의 7~22번째 위치를 0~15 배열에 저장한다. 아직 Base16 디코딩 전의 ASCII 문자 코드다. *)
                            KeyCode[KeyLength] := ChCode;
                            KeyLength := KeyLength + 1;
                        END_IF;

                        (* RSSI는 TAG 뒤 ;R + HEX 4자리로 전달된다. *)
                        IF BytePos = 23 THEN
                            RssiFieldOK := (ChCode = WORD#16#003B); (* ; *)
                        ELSIF BytePos = 24 THEN
                            RssiFieldOK := RssiFieldOK AND (ChCode = WORD#16#0052); (* R *)
                        ELSIF (BytePos >= 25) AND (BytePos <= 28) THEN
                            RssiAscii := WORD_TO_INT(ChCode);
                            (* HEXA는 대문자 A~F를 사용하므로 a~f는 먼저 대문자로 변환 *)
                            IF (RssiAscii >= INT#16#0061) AND (RssiAscii <= INT#16#0066) THEN
                                RssiAscii := RssiAscii - INT#16#0020;
                            END_IF;

                            RssiCharValid :=
                                ((RssiAscii >= INT#16#0030) AND (RssiAscii <= INT#16#0039))
                                OR ((RssiAscii >= INT#16#0041) AND (RssiAscii <= INT#16#0046));

                            IF RssiCharValid THEN
                                (* R12140~R12143에 F,E,1,A를 한 WORD당 한 문자씩 저장 *)
                                Z17 := BytePos - 25;
                                MOV(TRUE, RssiAscii, R12140Z17);
                                Z17 := 0;
                                RssiDigits := RssiDigits + 1;
                            ELSE
                                (* 잘못된 ASCII는 HEXA를 실행하지 않도록 차단 *)
                                RssiFieldOK := FALSE;
                            END_IF;

                        END_IF;

                    END_IF;

                    3:
                    (* CR 다음 LF이며 16바이트이면 완성 *)
                    KeyReady :=
                    (ChCode = WORD#16#000A)
                    AND (KeyLength = 16);
                    RssiMessageValid := KeyReady AND RssiFieldOK
                    AND (RssiDigits = 4) AND (BytePos = 28);
                    IF RssiMessageValid THEN
                        (* HEXA(실행조건, ASCII 시작주소, 문자 수, 변환 결과 시작주소)
                           SM8161=TRUE: 한 WORD의 하위 8비트에서 문자 하나씩 읽는다.
                           F,E,1,A -> 16#FE1A. 바이트 교환은 필요 없음.
                           출력은 R12144~R12147을 예약하고 결과 1WORD는 R12144에서 읽는다.
                           공용 모드 비트이므로 기존 상태를 저장하고 명령 직후 복원한다. *)
                        RssiModeSaved := SM8161;
                        SM8161 := TRUE;
                        RssiHexOk := HEXA(TRUE, R12140, 4, R12144);
                        SM8161 := RssiModeSaved;

                        RssiMessageValid := RssiHexOk;
                        IF RssiHexOk THEN
                            (* FE1A의 비트패턴을 signed INT -486으로 읽고 DINT로 확장 *)
                            RssiValue := INT_TO_DINT(WORD_TO_INT(R12144));
                        END_IF;
                    END_IF;

                    ParseState := 0;

                END_CASE;

            END_IF;


            //=============================================================================
            //2. '34464330' 포함 여부 검사, Hex 인코딩(Base16인코딩)방식임 h3334 ->string34로 표현됨
            //=============================================================================

            Match4FC0 := FALSE;

            IF KeyReady THEN

                (* 16문자 안에서 8문자 패턴 '34464330'을 찾는다. 현재는 모든 시작 위치를 검사하며 Base16 두 문자 경계 정렬은 별도로 검사하지 않는다. *)
                FOR SearchPos := 0 TO 8 DO

                    IF
                        (KeyCode[SearchPos]
                        = WORD#16#0033)
                        AND
                        (KeyCode[SearchPos + 1]
                        = WORD#16#0034)
                        AND
                        (KeyCode[SearchPos + 2]
                        = WORD#16#0034)
                        AND
                        (KeyCode[SearchPos + 3]
                        = WORD#16#0036)
                        AND
                        (KeyCode[SearchPos + 4]
                        = WORD#16#0034)
                        AND
                        (KeyCode[SearchPos + 5]
                        = WORD#16#0033)
                        AND
                        (KeyCode[SearchPos + 6]
                        = WORD#16#0033)
                        AND
                        (KeyCode[SearchPos + 7]
                        = WORD#16#0030)
                        THEN
                        //h3334343634333330 hex string

                        Match4FC0 := TRUE;
                        EXIT;

                    END_IF;

                END_FOR;

            END_IF;


            //=========================================
            //3. 조건에 맞는 Key만 집계
            //=========================================

            (* 34464330은 '4FC0'의 Base16 표현. 이 패턴을 포함하는 완성된 메시지만 집계한다. *)
            IF Match4FC0 THEN
                IF NOT RssiMessageValid THEN
                    WarnRssiData := TRUE;
                END_IF;

                (* -1은 일치하는 기존 TAG가 아직 없다는 표시. TAG 배열의 유효 인덱스는 0부터 시작한다. *)
                FoundIndex := -1;

                //기존 Key 검색
                IF DataTotal > 0 THEN

                    FOR jj := 0 TO DataTotal - 1 DO

                        SameKey := TRUE;

                        FOR kk := 0 TO 15 DO

                            (* 등록된 TAG의 16개 원본 문자 코드 중 하나라도 다르면 다른 TAG로 판단한다. *)
                            IF DataCode[jj, kk]<> KeyCode[kk] THEN

                                SameKey := FALSE;
                                EXIT;

                            END_IF;

                        END_FOR;

                        IF SameKey THEN

                            FoundIndex := jj;
                            EXIT;

                        END_IF;

                    END_FOR;

                END_IF;


                IF FoundIndex >= 0 THEN

                    //기존 Key: 횟수 증가
                    (* 같은 TAG의 응답 메시지 개수를 증가시킨다. 서로 다른 실제 물체 수를 뜻하지 않는다. *)
                    CountList[FoundIndex] := CountList[FoundIndex] + 1;

                ELSIF DataTotal < 3 THEN

                    //신규 Key: 등록
                    FOR kk := 0 TO 15 DO

                        DataCode[DataTotal, kk] := KeyCode[kk];

                    END_FOR;

                    (* 새 TAG의 첫 응답을 1회로 기록한다. 배열의 기존 값은 신규 등록 시 덮어쓴다. *)
                    CountList[DataTotal] := 1;
                    FoundIndex := DataTotal;
                    DataTotal := DataTotal + 1;

                ELSE

                    Overflow := TRUE;
                    (* 서로 다른 4번째 TAG 등록을 시도한 경고. 신규 TAG를 버리고 기존 3개의 집계는 계속한다. *)
                    ErrDataFull := TRUE;


                END_IF;
                (* 등록된 TAG별 최고 RSSI만 저장 *)
                IF (FoundIndex >= 0) AND RssiMessageValid THEN
                    IF (NOT RssiSeen[FoundIndex]) OR (RssiValue > RssiMax[FoundIndex]) THEN
                        RssiMax[FoundIndex] := RssiValue;
                        RssiSeen[FoundIndex] := TRUE;
                    END_IF;
                END_IF;


            END_IF;

        END_FOR;

        Z16 := 0;

        IF AnalyzeIndex >= RawCount THEN
            State := 60;
        END_IF;

    END_IF;


    //-----------------------------------------------------
    //60: TAG 디코딩 및 모드별 대표 TAG 선정
    //-----------------------------------------------------
    60:

    (* Base16 16문자 -> 모니터링용 ASCII 문자열 8문자 *)
    ResultText := '';
    IF DataTotal > 0 THEN
        FOR DecodeTag := 0 TO DataTotal - 1 DO
            TagText[DecodeTag] := '';
            TagTextError[DecodeTag] := FALSE;
            (* 원본 16문자를 두 문자씩 읽어 ASCII 8문자로 디코딩한다. *)
            FOR DecodePos := 0 TO 7 DO
                DecodeByte := 0;
                DecodePairOK := TRUE;
                FOR DecodePart := 0 TO 1 DO
                    (* DecodePart=0은 두 문자 중 앞 문자, 1은 뒤 문자. 예: 수신 문자 3과 4를 읽어 16진수 34를 만든다. *)
                    DecodeAscii := WORD_TO_INT(DataCode[DecodeTag, DecodePos * 2 + DecodePart]);
                    IF (DecodeAscii >= 48) AND (DecodeAscii <= 57) THEN
                        DecodeNibble := DecodeAscii - 48; (* '0'..'9' *)
                    ELSIF (DecodeAscii >= 65) AND (DecodeAscii <= 70) THEN
                        DecodeNibble := DecodeAscii - 55; (* 'A'..'F' *)
                    ELSIF (DecodeAscii >= 97) AND (DecodeAscii <= 102) THEN
                        DecodeNibble := DecodeAscii - 87; (* 'a'..'f' *)
                    ELSE
                        DecodeNibble := -1;
                        DecodePairOK := FALSE;
                    END_IF;
                    IF DecodeNibble >= 0 THEN
                        (* 앞 니블을 16배 한 뒤 뒤 니블을 더한다. 예: 3*16+4=52, ASCII 52는 문자 '4'다. *)
                        DecodeByte := DecodeByte * 16 + DecodeNibble;
                    END_IF;
                END_FOR;

                (* NUL/제어문자 때문에 문자열이 잘리지 않도록 '?'로 표시 *)
                (* 숫자와 영문자를 문자열 문자표에서 찾아 표시 *)
                IF DecodePairOK THEN
                    IF (DecodeByte >= 48) AND (DecodeByte <= 57) THEN
                        (* 0~9 *)
                        (* 여기부터 DecodeAscii는 원본 코드가 아니라 MID 문자표 위치로 재사용한다. 숫자 '0'의 ASCII 48을 위치 1로 바꾼다. *)
                        DecodeAscii := DecodeByte - 47;
                        (* MID 인수: 문자표, 꺼낼 문자 수 1, 시작 위치. 시작 위치는 1부터 센다. *)
                        DecodeChar := MID(
                        '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz',
                        1,
                        DecodeAscii
                        );

                    ELSIF (DecodeByte >= 65) AND (DecodeByte <= 90) THEN
                        (* A~Z *)
                        (* MID 문자표에서 대문자 A는 11번째. ASCII 65-54=11이다. *)
                        DecodeAscii := DecodeByte - 54;
                        (* MID 인수: 문자표, 꺼낼 문자 수 1, 시작 위치. 시작 위치는 1부터 센다. *)
                        DecodeChar := MID(
                        '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz',
                        1,
                        DecodeAscii
                        );

                    ELSIF (DecodeByte >= 97) AND (DecodeByte <= 122) THEN
                        (* a~z *)
                        (* MID 문자표에서 소문자 a는 37번째. ASCII 97-60=37이다. *)
                        DecodeAscii := DecodeByte - 60;
                        (* MID 인수: 문자표, 꺼낼 문자 수 1, 시작 위치. 시작 위치는 1부터 센다. *)
                        DecodeChar := MID(
                        '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz',
                        1,
                        DecodeAscii
                        );

                    ELSE
                        DecodeChar := '?';
                        TagTextError[DecodeTag] := TRUE;
                    END_IF;

                ELSE
                    DecodeChar := '?';
                    TagTextError[DecodeTag] := TRUE;
                END_IF;
                (* 이번에 변환한 한 문자를 이전 문자열 끝에 붙인다. 8회 반복하면 8문자가 된다. *)
                TagText[DecodeTag] := CONCAT(TagText[DecodeTag], DecodeChar);

                (* TAG 8문자를 종료값 없이 4워드로 저장. 표시 문자열과 동일하게 오류 문자는 '?' *)
                IF (NOT DecodePairOK) OR NOT (
                    ((DecodeByte >= 48) AND (DecodeByte <= 57))
                    OR ((DecodeByte >= 65) AND (DecodeByte <= 90))
                    OR ((DecodeByte >= 97) AND (DecodeByte <= 122))) THEN
                    (* ASCII 63은 '?'. 잘못된 Base16 또는 숫자/영문 이외 문자는 물음표로 저장한다. *)
                    DecodeByte := 63;
                END_IF;
                (* TAG당 6워드 간격 + 현재 문자의 WORD 위치. 문자 0,1은 첫 WORD / 2,3은 다음 WORD에 들어간다. *)
                Z17 := DecodeTag * 6 + DecodePos / 2;
                IF (DecodePos MOD 2) = 0 THEN
                    (* 짝수 문자: 숫자 값을 WORD 자료형으로 바꾸어 하위 8비트에 저장하고 상위 8비트를 0으로 만든다. *)
                    WorkWord := INT_TO_WORD(DecodeByte);
                ELSE
                    (* 홀수 문자: 현재 문자를 상위 8비트로 옮겨 앞 문자와 합친다. 예: '4'(0034) + 'F'(4600) -> 4634. *)
                    WorkWord := R12100Z17 OR SHL(INT_TO_WORD(DecodeByte), 8);
                END_IF;
                (* 조합 중인 16비트 WORD를 R영역에 전송한다. 별도 NUL 종료 WORD는 쓰지 않는다. *)
                MOV(TRUE, WorkWord, R12100Z17);

            END_FOR;
        END_FOR;
    END_IF;


    (* 반복횟수 공개: TAG당 6워드 = TAG 4워드 + DINT 횟수 2워드 *)
    (* [0] R12100~R12103: TAG / R12104~R12105: 횟수 *)
    (* [1] R12106~R12109: TAG / R12110~R12111: 횟수 *)
    (* [2] R12112~R12115: TAG / R12116~R12117: 횟수 *)
    (* R12118: 등록 TAG 개수 *)
    IF DataTotal > 0 THEN
        FOR DecodeTag := 0 TO DataTotal - 1 DO
            Z17 := DecodeTag * 6;
            (* DINT 횟수는 32비트이므로 연속 2워드를 전송한다. 첫 주소는 하위 WORD, 다음 주소는 상위 WORD다. *)
            DMOV(TRUE, CountList[DecodeTag], R12104Z17);
        END_FOR;
    END_IF;
    Z17 := 0;
    R12118 := DataTotal;

    (* 1. 모든 모드에서 최다 COUNT와 해당 TAG 위치 확인 *)
    IF DataTotal > 0 THEN
        FOR jj := 0 TO DataTotal - 1 DO
            IF CountList[jj] > MaxCount THEN
                MaxCount := CountList[jj];
                CountBestIndex := jj;
            END_IF;
        END_FOR;
    END_IF;
    BestIndex := CountBestIndex;

    (* 2. RSSI 모드는 전체 TAG, 혼합 모드는 COUNT 차이가 설정값 이하인 TAG 비교 *)
    IF (ActiveSelectMode <> 0) AND (DataTotal > 0) THEN
        FOR jj := 0 TO DataTotal - 1 DO
            IF (ActiveSelectMode = 1)
                OR ((MaxCount - CountList[jj]) <= CountGapLimit) THEN
                IF RssiSeen[jj] THEN
                    IF RssiBestIndex < 0 THEN
                        RssiBestIndex := jj;
                    ELSIF RssiMax[jj] > RssiMax[RssiBestIndex] THEN
                        RssiBestIndex := jj;
                    END_IF;
                ELSE
                    CandidatesValid := FALSE;
                END_IF;
            END_IF;
        END_FOR;

        IF ActiveSelectMode = 1 THEN
            BestIndex := RssiBestIndex;
            IF BestIndex < 0 THEN
                ErrRssiMissing := TRUE;
                Error := TRUE;
            END_IF;
        ELSIF CandidatesValid THEN
            BestIndex := RssiBestIndex;
        ELSE
            (* 혼합 후보 중 RSSI 누락 시 최다 COUNT 결과 사용 *)
            HybridFallback := TRUE;
        END_IF;
    END_IF;

    (* 3. 모니터링용 TAG별 최고 RSSI와 유효 여부 출력 *)
    FOR jj := 0 TO 2 DO
        Z17 := jj;
        MOV(TRUE, DINT_TO_INT(RssiMax[jj]), R12126Z17);
        R12130Z17 := 0;
        IF RssiSeen[jj] THEN R12130Z17 := 1; END_IF;
    END_FOR;
    Z17 := 0;

    IF BestIndex >= 0 THEN
        ResultText := TagText[BestIndex];
        ResultCount := CountList[BestIndex];
        ResultRssiValid := RssiSeen[BestIndex];
        IF ResultRssiValid THEN
            ResultRssi := RssiMax[BestIndex];
        END_IF;
        MOV(TRUE, DINT_TO_INT(ResultRssi), R12129);

        (* 선정 TAG 별도 저장: 8문자 4워드, 종료값 없음 *)
        FOR kk := 0 TO 3 DO
            (* 선정 TAG가 저장된 R12100 기준 오프셋. kk=0~3으로 해당 TAG의 4워드를 읽는다. *)
            Z15 := BestIndex * 6 + kk;
            Z17 := kk;
            (* 선정 TAG의 4워드를 R12120~R12123에 별도로 복사한다. 원래 TAG1~3의 순서는 유지한다. *)
            MOV(TRUE, R12100Z15, R12120Z17);
        END_FOR;
        Z15 := 0;
        Z17 := 0;
        (* 반복횟수는 DINT이므로 2워드 전송 *)
        DMOV(TRUE, ResultCount, R12124);



        (* 기존 R12801 원본 출력 사용 안 함
        FOR kk := 0 TO 15 DO

        ResultCode[kk] := DataCode[BestIndex, kk];
        Z17 := kk;
        R12801Z17 := ResultCode[kk];

        END_FOR;

        Z17 := 0;
        *)

    END_IF;

    ResultValid :=
    (BestIndex >= 0) AND NOT Error;

    State := 90;


    //-----------------------------------------------------
    //90: 처리 완료
    //-----------------------------------------------------
    90:

    (* 분석 및 결과 저장 후 D4000~D5999만 지운다. R12100 이후의 결과는 다음 Start까지 유지한다. *)
    FMOV(TRUE,0,K2000,D4000);//종료후 D4000~D5999 삭제, 시작할때 RawCount개수만 읽지만 삭제한다.

    IF Error THEN
        ResultValid := FALSE;
    END_IF;

    (* 조건에 맞는 TAG가 없고 처리 오류가 없을 때 미인식 *)
    (* TAG 미인식과 처리 오류를 구분한다. 사용자가 Stop하여 TAG 없이 종료한 경우에도 이 조건이면 TRUE다. *)
    TagReadFail := (DataTotal = 0) AND NOT Error;
    (* 완료 표시를 남긴 뒤 State를 0으로 복귀한다. State 0에서 매 스캔 Done을 지우지 않으므로 다음 Start까지 유지된다. *)
    Done := TRUE;
    State := 0;


    //-----------------------------------------------------
    //잘못된 상태값
    //-----------------------------------------------------
ELSE
    ErrState := TRUE;
    Error := TRUE;
    ResultValid := FALSE;
    State := 90;

END_CASE;

// 원인별 오류는 다음 측정 시작까지 유지. 매 스캔 FALSE로 초기화하지 않음.
(* 원인별 오류 BOOL을 합산한다. 각 원인은 Start에서만 초기화한다. 중간 Error:=TRUE는 같은 스캔의 분기 판단을 위한 것이다. *)
Error :=
ErrComm                 (* 통신 이상 신호 CommError 발생 *)
OR ErrInventoryTimeout  (* Inventory 송신 완료 대기 중 6초 초과 *)
OR ErrInventorySend     (* Inventory 송신 명령 이상 완료 *)
OR ErrReceive           (* 수집·정지 처리 중 수신 명령 이상 완료 *)
OR ErrStopTimeout       (* Stop 송신 완료 대기 중 6초 초과 *)
OR ErrStopSend          (* Stop 송신 명령 이상 완료 *)
OR ErrRawCount          (* 누적 수신 바이트 수 RawCount가 0 미만 또는 2000 초과 *)
OR ErrRxLength          (* 수신 길이 RxLength가 0 미만 또는 2046 초과 *)
OR ErrAnalyzeIndex      (* 분석 위치 AnalyzeIndex가 0 미만 또는 RawCount 초과 *)
OR ErrSelectMode       (* START 시 R12119가 0~2 범위 밖 *)
OR ErrRssiMissing      (* RSSI 모드에서 등록 TAG는 있으나 유효 RSSI 후보가 없음 *)
OR ErrState;            (* State가 CASE에 정의되지 않은 값일 때 *)
(* 용량 초과는 경고로 유지. Error 및 ResultValid를 무효화하지 않음 *)
(* ErrBufferFull: 2000바이트 누적 용량 초과로 추가 수신 중지 *)
(* ErrDataFull: 서로 다른 4번째 TAG는 제외하고 등록된 3개만 집계 *)
Overflow := ErrBufferFull OR ErrDataFull;
IF Error THEN
    ResultValid := FALSE;
END_IF;

(* 작업 대기 상태 0이면 FALSE. 완료 스캔에서 State가 0으로 복귀하므로 Done=TRUE, Busy=FALSE가 된다. *)
Busy := (State <> 0);


(* 래더에서 이 값으로 송신 버퍼/명령 종류를 선택한다. 실제 송신 실행은 TxRequest 신호로 요청한다. *)
D3000 := TxCommand;
(* 외부 송신 래더가 읽을 한 스캔 요청 비트. 동일 비트를 다른 곳에서 덮어쓰지 않도록 한다. *)
D3001.0 := TxRequest;

(* 정상 결과 완료 출력. 다음 Start까지 유지 *)
(* 작업 완료 AND 유효 결과일 때 ON. Overflow 경고가 있어도 다른 Error가 없으면 ON 가능하다. *)
D3001.1 := Done AND ResultValid;

(* 외부 래더/HMI용 종합 오류 비트. 버퍼/등록 개수 초과 경고는 현재 Error 집계에서 제외돼 있다. *)
D3001.2 := Error;

R12133 := 0;
IF ResultRssiValid THEN R12133 := 1; END_IF;
R12134 := 0;
IF WarnRssiData THEN R12134 := 1; END_IF;
R12135 := 0;
IF ErrRssiMissing THEN R12135 := 1; END_IF;
R12136 := 0;

MOV(TRUE, ActiveSelectMode, R12137);

R12139 := 0;
